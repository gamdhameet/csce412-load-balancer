58void validate(const Config& config) {
59 if (config.min_task_cycles < 1 || config.max_task_cycles < config.min_task_cycles)
60 throw std::invalid_argument("task cycle range must be positive and ordered");
61 for (const double p : {config.arrival_probability,
62 config.blocked_source_probability, config.processing_fraction}) {
63 if (!(p >= 0.0 && p <= 1.0))
64 throw std::invalid_argument("probabilities must be between 0 and 1");
65 }
66 if (config.scale_interval == 0 || config.min_servers_per_pool == 0 ||
67 config.max_servers_per_pool < config.min_servers_per_pool ||
68 config.max_queue_per_pool == 0 || config.max_servers_per_pool > SIZE_MAX / 80) {
69 throw std::invalid_argument("scale interval, server bounds, and queue limit must be positive");
70 }
71 IpFilter filter;
72 for (const auto& cidr : config.blocked_cidrs) filter.add_range(cidr);
73}

77Config load_config(const std::string& path) {
78 std::ifstream file(path);
79 if (!file) throw std::runtime_error("cannot open configuration: " + path);
80 Config config;
81 std::set<std::string> seen;
82 std::string line;
83 unsigned line_number = 0;
84 while (std::getline(file, line)) {
85 ++line_number;
86 line = trim(line.substr(0, line.find('#')));
87 if (line.empty()) continue;
88 const auto equal = line.find('=');
89 if (equal == std::string::npos)
90 throw std::invalid_argument("config line " + std::to_string(line_number) + " needs =");
91 const std::string key = trim(line.substr(0, equal));
92 const std::string value = trim(line.substr(equal + 1));
93 if (key.empty() || value.empty())
94 throw std::invalid_argument("empty key or value on config line " + std::to_string(line_number));
95 if (key != "blocked_cidr" && !seen.insert(key).second)
96 throw std::invalid_argument("duplicate config key: " + key);
97
98 if (key == "min_task_cycles") config.min_task_cycles = number<int>(value, key);
99 else if (key == "max_task_cycles") config.max_task_cycles = number<int>(value, key);
100 else if (key == "arrival_probability") config.arrival_probability = number<double>(value, key);
101 else if (key == "blocked_source_probability") config.blocked_source_probability = number<double>(value, key);
102 else if (key == "processing_fraction") config.processing_fraction = number<double>(value, key);
103 else if (key == "scale_interval") config.scale_interval = number<unsigned>(value, key);
104 else if (key == "min_servers_per_pool") config.min_servers_per_pool = number<std::size_t>(value, key);
105 else if (key == "max_servers_per_pool") config.max_servers_per_pool = number<std::size_t>(value, key);
106 else if (key == "max_queue_per_pool") config.max_queue_per_pool = number<std::size_t>(value, key);
107 else if (key == "blocked_cidr") config.blocked_cidrs.push_back(value);
108 else throw std::invalid_argument("unknown config key: " + key);
109 }
110 validate(config);
111 return config;
112}
