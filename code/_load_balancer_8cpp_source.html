38void LoadBalancer::tick(unsigned cycle, std::ostream* verbose) {
39 for (auto& server : servers_) {
40 if (server.tick()) {
41 ++stats_.completed;
42 if (verbose) *verbose << "cycle " << cycle << " " << name_
43 << " complete server=" << server.id() << '\n';
44 }
45 }
46 for (auto& server : servers_) {
47 if (queue_.empty()) break;
48 if (!server.busy()) {
49 const Request request = queue_.front();
50 queue_.pop();
51 server.assign(request);
52 ++stats_.assigned;
53 if (verbose) *verbose << "cycle " << cycle << " " << name_
54 << " assign server=" << server.id()
55 << " source=" << request.ip_in
56 << " time=" << request.time
57 << " waiting=" << queue_.size() << '\n';
58 }
59 }
60 if (cycle % scale_interval_ == 0) scale(cycle, verbose);
61}

63void LoadBalancer::scale(unsigned cycle, std::ostream* verbose) {
64 const std::size_t count = servers_.size();
65 std::string action;
66 if (queue_.size() > 80 * count && count < max_servers_) {
67 servers_.emplace_back(next_id_++);
68 ++stats_.servers_added;
69 stats_.peak_servers = std::max(stats_.peak_servers, servers_.size());
70 action = "add";
71 } else if (queue_.size() < 50 * count && count > min_servers_) {
72 const auto idle = std::find_if(servers_.begin(), servers_.end(),
73 [](const WebServer& server) { return !server.busy(); });
74 if (idle != servers_.end()) {
75 servers_.erase(idle);
76 ++stats_.servers_removed;
77 stats_.minimum_servers = std::min(stats_.minimum_servers, servers_.size());
78 action = "remove";
79 }
80 }
81 if (!action.empty()) {
82 scale_events_.push_back({cycle, action, servers_.size(), queue_.size()});
83 if (verbose) *verbose << "cycle " << cycle << " " << name_
84 << " scale=" << action << " servers=" << servers_.size()
85 << " waiting=" << queue_.size() << '\n';
86 }
