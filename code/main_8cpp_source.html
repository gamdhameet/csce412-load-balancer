41unsigned long long positive_integer(const std::string& text, const std::string& label) {
42 if (text.empty() || text[0] == '-' || text.find_first_not_of("0123456789") != std::string::npos)
43 throw std::invalid_argument(label + " must be a positive integer");
44 std::size_t used = 0;
45 const unsigned long long value = std::stoull(text, &used);
46 if (used != text.size() || value == 0)
47 throw std::invalid_argument(label + " must be a positive integer");
48 return value;
49}

51Options parse_options(int argc, char* argv[]) {
52 Options options;
53 for (int i = 1; i < argc; ++i) {
54 const std::string key = argv[i];
55 if (key == "--help") {
56 std::cout << "Usage: ./load-balancer [--servers N] [--cycles N] [--seed N] "
57 "[--mode quiet|verbose] [--config FILE] [--log FILE]\n"
58 "Missing server count or cycle count is requested interactively.\n";
59 std::exit(0);
60 }
61 if (i + 1 >= argc) throw std::invalid_argument("missing value after " + key);
62 const std::string value = argv[++i];
63 if (key == "--servers") options.servers = positive_integer(value, "servers");
64 else if (key == "--cycles") {
65 const auto n = positive_integer(value, "cycles");
66 if (n >= std::numeric_limits<unsigned>::max()) throw std::invalid_argument("cycles too large");
67 options.cycles = static_cast<unsigned>(n);
68 } else if (key == "--seed") {
69 if (value.empty() || value[0] == '-' || value.find_first_not_of("0123456789") != std::string::npos)
70 throw std::invalid_argument("seed must be a nonnegative integer");
71 const auto n = std::stoull(value);
72 if (n > UINT32_MAX) throw std::invalid_argument("seed too large");
73 options.seed = static_cast<std::uint32_t>(n);
74 } else if (key == "--mode") options.mode = value;
75 else if (key == "--config") options.config_path = value;
76 else if (key == "--log") options.log_path = value;
77 else throw std::invalid_argument("unknown option: " + key);
78 }
79 if (options.mode != "quiet" && options.mode != "verbose")
80 throw std::invalid_argument("mode must be quiet or verbose");
81 return options;
82}

149int main(int argc, char* argv[]) {
150 try {
151 Options options = parse_options(argc, argv);
152 const Config config = load_config(options.config_path);
153 prompt_if_missing(options);
154 if (options.servers < 2 * config.min_servers_per_pool ||
155 (options.servers + 1) / 2 > config.max_servers_per_pool ||
156 options.servers / 2 > config.max_servers_per_pool)
157 throw std::invalid_argument("initial server count is outside the two pool limits");
158
159 IpFilter filter;
160 for (const auto& cidr : config.blocked_cidrs) filter.add_range(cidr);
161 Switch router(options.servers, config.min_servers_per_pool,
162 config.max_servers_per_pool, config.max_queue_per_pool,
163 config.scale_interval, filter);
164 RequestGenerator generator(options.seed, config.min_task_cycles, config.max_task_cycles,
165 config.processing_fraction, config.blocked_source_probability);
166
167 const std::size_t target_processing = ((options.servers + 1) / 2) * 100;
168 const std::size_t target_streaming = (options.servers / 2) * 100;
169 if (target_processing > config.max_queue_per_pool || target_streaming > config.max_queue_per_pool)
170 throw std::invalid_argument("max_queue_per_pool is too small for the initial queue");
171
172 std::ofstream log_file;
173 if (!options.log_path.empty()) {
174 const auto parent = std::filesystem::path(options.log_path).parent_path();
175 if (!parent.empty()) std::filesystem::create_directories(parent);
176 log_file.open(options.log_path);
177 if (!log_file) throw std::runtime_error("cannot write log: " + options.log_path);
178 }
179 std::ostream* verbose = options.mode == "verbose"
180 ? (log_file ? static_cast<std::ostream*>(&log_file) : &std::cout) : nullptr;
181
182 // Seed each queue to 100 requests per initial server. Blocked attempts are replaced.
183 const std::size_t max_initial_attempts = 100 * (target_processing + target_streaming);
184 while (router.processing().waiting() < target_processing) {
185 if (router.generated_count() >= max_initial_attempts)
186 throw std::runtime_error("could not fill initial queue; check blocked source settings");
187 Request request = generator.next();
188 request.type = 'P';
189 router.route(request, 0, nullptr);
190 }
191 while (router.streaming().waiting() < target_streaming) {
192 if (router.generated_count() >= max_initial_attempts)
193 throw std::runtime_error("could not fill initial queue; check blocked source settings");
194 Request request = generator.next();
195 request.type = 'S';
196 router.route(request, 0, nullptr);
197 }
198 const std::size_t starting_queue = router.waiting();
199 const std::size_t initial_generated = router.generated_count();
200 std::ostringstream start;
201 start << "CodeHarbor load balancer | start\n"
202 << "Initial servers: " << options.servers << " | clock cycles: " << options.cycles
203 << " | starting queue size: " << starting_queue << '\n'
204 << "Task time range: " << config.min_task_cycles << "-" << config.max_task_cycles
205 << " cycles | blocked CIDR ranges: " << filter.range_count() << "\n\n";
206 print_terminal_block(start.str());
207 if (log_file) log_file << start.str();
208 if (verbose) *verbose << "cycle 0 initial queues loaded: processing="
209 << router.processing().waiting() << " streaming="
210 << router.streaming().waiting() << '\n';
211 for (unsigned cycle = 1; cycle <= options.cycles; ++cycle) {
212 if (generator.arrives(config.arrival_probability))
213 router.route(generator.next(), cycle, verbose);
214 router.tick(cycle, verbose);
215 }
216 const std::string report = summary(router, options, config, starting_queue, initial_generated);
217 print_terminal_block(report);
218 if (log_file) log_file << report;
219 return 0;
220 } catch (const std::exception& error) {
221 std::cerr << "error: " << error.what() << '\n';
222 return 1;
223 }
